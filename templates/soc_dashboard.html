    async function updateDashboardStats() {
        try {
            const res = await fetch('/api/dashboard/stats'); const data = await res.json();

                const s = data.stats;
                updateElement('totalLogs', s.total_logs);
                updateElement('activeAlerts', s.active_alerts);
                updateElement('falsePositives', s.false_positives);
                updateElement('openIncidents', s.open_incidents);
                updateElement('resolvedIncidents', s.resolved_incidents);
                updateElement('agentCount', s.active_agents || 0);
                
                // Security Posture
                const threatScore = data.threat_score || 0;
                const accuracy = data.detection_accuracy || 0;
                updateElement('stat-threat-score', threatScore + '%');
                updateElement('stat-accuracy', accuracy + '%');
                document.getElementById('bar-threat-score').style.width = threatScore + '%';
                document.getElementById('bar-accuracy').style.width = accuracy + '%';

                renderTimelineChart(data.logs_per_hour || {});
                renderMitreTacticMap(data.mitre_counts || {});
                renderAlertTable(data.recent_malicious || []);
                renderSeverityChart(data.severity_breakdown || {});
                renderAgentStatus(data.agents || []);
            }
        } catch (e) { console.error(e); }
    }

    function renderTimelineChart(data) {

    async function performAction(action, params = {}, successMsg = "Action completed") {
        if (!currentIncidentId) return;
        try {
            const res = await fetch(`/api/incidents/${currentIncidentId}/action`, {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({ action, params })
            });
            const data = await res.json();
            if (data.status === 'success') {
                showToast(successMsg, "success");
                loadIncidentDetails(currentIncidentId); // Refresh UI and timeline
                return data;
            } else {
                showToast(data.message || "Action failed", "error");
            }
        } catch (e) {
            console.error(e);
            showToast("Error connecting to server", "error");
        }
    }

    async function loadIncidentQueue() {
        try {
            const res = await fetch('/api/incidents'); const data = await res.json();
            if (data.status === 'success') {
                const list = document.getElementById('incident-queue-list');
                const activeIncidents = data.incidents.filter(inc => inc.status !== 'Resolved');
                
                if (activeIncidents.length === 0) {
                    list.innerHTML = '
No active incidents.
';
                    document.getElementById('queue-count').textContent = '0';
                    return;
                }

                list.innerHTML = activeIncidents.map(inc => `
                    

                        

                            INC-${inc.id}
                            ${inc.severity}
                        

                        
${inc.title}

                        
${inc.status} · ${inc.owner || 'Unassigned'}

                    

                `).join('');
                document.getElementById('queue-count').textContent = activeIncidents.length;
                if (!currentIncidentId && activeIncidents.length > 0) loadIncidentDetails(activeIncidents[0].id);
            }
        } catch (e) { console.error(e); }
    }

    async function loadIncidentDetails(id) {
        currentIncidentId = id;
        document.querySelectorAll('.queue-item').forEach(item => item.classList.remove('active'));
        const activeItem = Array.from(document.querySelectorAll('.queue-item')).find(item => item.innerHTML.includes('INC-' + id));
        if (activeItem) activeItem.classList.add('active');

        try {
            const res = await fetch(`/api/incidents/${id}`); const data = await res.json();
            if (data.status === 'success') {
                const inc = data.data.incident;
                updateElement('triage-incident-id', `INC-${inc.id}`);
                updateElement('triage-status', inc.status);
                updateElement('triage-confidence', `${Math.round((inc.confidence || 0.92) * 100)}%`);
                updateElement('triage-assigned', inc.owner || 'Unassigned');
                updateElement('triage-host', inc.host);
                updateElement('triage-user', inc.user);
                updateElement('triage-mitre', inc.mitre.length > 0 ? inc.mitre.map(m => `${m.id}: ${m.name}`).join(' | ') : '--');
                
                const summaryLines = inc.executive_summary.split('\n\n');
                document.getElementById('triage-summary-display').innerHTML = summaryLines.map(line => `
${line.replace(/\n/g, '
')}
`).join('');

                renderProcessTree(inc.process_tree);
                renderTimeline(inc.timeline);
                document.getElementById('raw-evidence-display').textContent = JSON.stringify(inc.raw_evidence, null, 2);
                
                document.getElementById('detection-logic-panel').innerHTML = `
Source: ${inc.detection_logic || 'N/A'}
Method: Behavioral + Signature Analysis
Trigger: Detected malicious system commands via telemetry flow.
`;
                
                const iocList = document.getElementById('ioc-panel-display');
                if (inc.iocs && inc.iocs.length > 0) iocList.innerHTML = inc.iocs.map(ioc => `
${ioc.type}${ioc.value}
${ioc.risk}
`).join('');
                else iocList.innerHTML = '
No indicators identified
';

                const notesList = document.getElementById('notes-display');
                notesList.innerHTML = data.notes.map(n => `
${n.author}${n.created_at}
${n.note}
`).join('') || '
No notes yet.
';
            }
        } catch (e) { console.error(e); }
    }

    function renderProcessTree(tree) {
        const container = document.getElementById('process-tree-display');
        if (!tree || !tree.name) { container.innerHTML = '
No lineage data available
'; return; }
        
        function buildNode(node) {
            const children = node.children && node.children.length > 0 
                ? node.children.map(child => buildNode(child)).join('') 
                : '';
            
            return `
                

                    

                        

                            ${node.name}
                            PID: ${node.pid || 'N/A'}
                        

                        ${node.command || 'N/A'}
                    

                    ${children}
                

            `;
        }
        container.innerHTML = `
${buildNode(tree)}
`;
    }

    function renderTimeline(timeline) {
        const container = document.getElementById('timeline-display');
        if (!timeline || timeline.length === 0) { container.innerHTML = 'No timeline available'; return; }
        container.innerHTML = timeline.map(item => `
${item.timestamp}
${item.type}
${item.description}
`).join('');
    }

    async function updateIncidentStatus(status) {
        if (!currentIncidentId) return;
        if (status === 'Resolved' && !confirm("Close this incident and mark as Resolved?")) return;
        
        try {
            const res = await fetch(`/api/incidents/${currentIncidentId}/status`, {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({ status: status })
            });
            const data = await res.json();
            if (data.status === 'success') {
                showToast(`Incident status updated to ${status}`, "success");
                if (status === 'Resolved') {
                    loadIncidentQueue();
                } else {
                    loadIncidentDetails(currentIncidentId);
                }
            }
        } catch (e) { console.error(e); }
    }
